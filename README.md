        /* =========================================================
           4. ĐIỀU KHIỂN 2 BANNER TỰ TRƯỢT + HỖ TRỢ VUỐT MOBILE
        ========================================================== */
        var currentSlide = 0;
        var totalSlides = 2;
        var sliderTimer = null;

        function updateSliderUI() {
            var track = document.getElementById('hero-track');
            if (!track) return;
            track.style.transform = 'translateX(-' + (currentSlide * 100) + '%)';

            for (var i = 0; i < totalSlides; i++) {
                var dot = document.getElementById('dot-' + i);
                if (!dot) continue;
                if (i === currentSlide) {
                    dot.className = 'w-7 h-2 rounded-full bg-brand-accent transition-all';
                } else {
                    dot.className = 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
                }
            }
        }

        function nextSlide() {
            currentSlide = (currentSlide + 1) % totalSlides;
            updateSliderUI();
            resetSliderAuto();
        }

        function prevSlide() {
            currentSlide = (currentSlide - 1 + totalSlides) % totalSlides;
            updateSliderUI();
            resetSliderAuto();
        }

        function goToSlide(index) {
            currentSlide = index;
            updateSliderUI();
            resetSliderAuto();
        }

        function resetSliderAuto() {
            if (sliderTimer) clearInterval(sliderTimer);
            sliderTimer = setInterval(function() {
                currentSlide = (currentSlide + 1) % totalSlides;
                updateSliderUI();
            }, 5500);
        }

        /* Hỗ trợ vuốt cảm ứng trên điện thoại cho Slider */
        function initSliderTouch() {
            var slider = document.getElementById('hero-slider');
            if (!slider) return;
            var startX = 0;
            slider.addEventListener('touchstart', function(e) {
                startX = e.touches[0].clientX;
            }, { passive: true });
            slider.addEventListener('touchend', function(e) {
                var endX = e.changedTouches[0].clientX;
                var diff = startX - endX;
                if (Math.abs(diff) > 45) {
                    if (diff > 0) nextSlide();
                    else prevSlide();
                }
            }, { passive: true });
        }

        function submitQuoteForm(e) {
            e.preventDefault();
            var name = document.getElementById('f-name').value.trim();
            var phone = document.getElementById('f-phone').value.trim();
            var type = document.getElementById('f-type').value;
            var budget = document.getElementById('f-budget').value;
            var feedback = document.getElementById('form-feedback');

            if (!name || !phone) {
                feedback.className = 'mt-3 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-xs sm:text-sm font-medium';
                feedback.textContent = 'Vui lòng nhập đầy đủ Họ tên và Số điện thoại để Phú Minh Đạt liên hệ lại.';
                return;
            }

            feedback.className = 'mt-3 p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-800 text-xs sm:text-sm font-medium';
            feedback.innerHTML = 'Cảm ơn <strong>' + name + '</strong>! Yêu cầu báo giá hạng mục <strong>' + type + '</strong> (ngân sách: ' + budget + ') đã được ghi nhận. Kỹ sư Phú Minh Đạt sẽ liên hệ số <strong>' + phone + '</strong> trong thời gian sớm nhất.';
            document.getElementById('quote-form').reset();
        }
